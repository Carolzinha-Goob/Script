-- LocalScript (StarterPlayerScripts)

local Players = game:GetService("Players")
local player = Players.LocalPlayer

-- Cria a aba
local TpPlayers = Window:CreateTab("TP to Player", 4483362458)

-- Tabela para guardar os botões criados
local playerButtons = {}

-- Função para criar botão de teleporte
local function createPlayerButton(plr)
    local Button = TpPlayers:CreateButton({
        Name = plr.Name,
        Callback = function()
            if player.Character and player.Character:FindFirstChild("HumanoidRootPart") 
            and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
                player.Character.HumanoidRootPart.CFrame = plr.Character.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
            end
        end,
    })

    -- Guarda referência
    playerButtons[plr] = Button
end

-- Cria botões para jogadores já presentes
for _, plr in pairs(Players:GetPlayers()) do
    if plr ~= player then
        createPlayerButton(plr)
    end
end

-- Atualiza lista quando novos jogadores entram
Players.PlayerAdded:Connect(function(plr)
    if plr ~= player then
        createPlayerButton(plr)
    end
end)

-- Remove botão quando jogador sai
Players.PlayerRemoving:Connect(function(plr)
    if playerButtons[plr] then
        playerButtons[plr]:Destroy() -- remove o botão da aba
        playerButtons[plr] = nil
    end
end)
